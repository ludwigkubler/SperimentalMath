def generate_d_regular_graph(n, d):
    if n * d % 2 != 0 or d > n - 1:

    graph = [[] for _ in range(n)]
    degree_count = [0] * n
    
    def add_edge(u, v):
        if u == v or len(graph[u]) >= d or len(graph[v]) >= d:
            return False
        graph[u].append(v)
        graph[v].append(u)
        degree_count[u] += 1
        degree_count[v] += 1
        return True

def lcm(a, b):
