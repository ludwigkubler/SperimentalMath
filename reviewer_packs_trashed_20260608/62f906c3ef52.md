def generate_xor_function(n):

def fast_walsh_hadamard_transform(f):
    n = len(f)
    while n > 1:
        half_n = n // 2
        for i in range(half_n):
            for j in range(n):
                if j < half_n:
                    f[j] += f[j + half_n]
                else:
                    f[j] -= f[j - half_n]
        n //= 2
    return [x / (1 << len(f)) for x in f]

def elementary_symmetric_polynomial_expansion(coeffs):
    n = len(coeffs)

        C[0] -= coeff
