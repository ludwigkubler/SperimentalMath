def run_trial(seed: int) -> dict:

        if n < 3:
            return True

            for j in range(i + 1, n):
                for k in range(j + 1, n):
                    if (i * j + j * k + k * i) % (n - 1) == 0:
                        return False
        return True

                    edges.add((u, v))
            return list(edges)
