        random.shuffle(clique)
        return clique[:min(2 * int(math.log2(n)), len(clique))]
    
    def free_group_from_clique(clique):
        generators = set()
        relations = []
        for u, v in clique:
            generators.add(f'x{u}')
            generators.add(f'x{v}')
            relations.append((f'x{u}', f'x{v}'))
