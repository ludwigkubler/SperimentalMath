def generate_circuit(n, s):

            if _ % 2 == 0:
                gate_type = 'AND'
            else:
                gate_type = 'OR'
            layer.append(gate_type)
        circuit.extend(layer)
    return circuit

def simulate_circuit(circuit, n):
    stack = []
    for gate in circuit:
        if gate == 'NOT':
            a = stack.pop()
            stack.append(not a)
        elif gate == 'AND':
            b, a = stack.pop(), stack.pop()
            stack.append(a and b)
        elif gate == 'OR':
            b, a = stack.pop(), stack.pop()
            stack.append(a or b)
    return stack[0]

def estimate_bias(circuit, n):
    N = 2048
    count = 0
    for _ in range(N):

        if result % 3 == 1:

    return bias
