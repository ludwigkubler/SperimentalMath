        return bp
    
    def calculate_transition_matrix(bp):
        n = len(bp)
        T = [[0] * (n + 1) for _ in range(n + 1)]
        for i in range(n):
            for j in range(n):
                if bp[i][j % 2] == 1:
                    T[i][j + 1] += 1
                else:
                    T[i][j] += 1
        return T

        moments = [0] * (k + 1)
        for i in range(len(T)):
            for j in range(len(T[0])):
                moments[0] += T[i][j]
                if i != j:
                    moments[1] += abs(i - j) * T[i][j]
        return moments

        if n < 2:
            return 0

        for k in range(1, n):

        return rho
