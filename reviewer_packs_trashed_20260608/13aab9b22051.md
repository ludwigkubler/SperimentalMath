def run_trial(seed: int) -> dict:

    def generate_polynomial(n, w):
        # Generate a random polynomial over GF(2)
        poly = [random.choice([0, 1]) for _ in range(n)]
        return poly
    
    def partial_derivatives(poly_str, vars, x_values):
        # Compute partial derivatives
        partials = []
        for var in vars:
            partial = ""
            for i, coeff in enumerate(poly_str.split()):
                if var in coeff:
                    power = int(coeff[coeff.index(var) + 1:])
                    if power > 0:
                        new_coeff = str((power * int(coeff[:coeff.index(var)])) % 2)
                        new_var = f"{var}^{power-1}"
                        partial += f"{new_coeff}*{new_var} "
            partials.append(partial.strip())
        return partials
    
    def construct_ideal(partials):
        # Construct the ideal generated by partial derivatives
        I = set()
        for part in partials:
            I.update(set(part.split()))
        return I
    
    def castelnuovo_mumford_regularity(I, n):
        # Compute Castelnuovo-Mumford regularity (simplified)
        return len(I) * math.log(n, 2)
    
    n = random.randint(5, 40)
    w = random.randint(1, 10)
    poly_str = generate_polynomial(n, w)
    vars = [f"x{i}" for i in range(w)]
    x_values = [random.randint(0, 1) for _ in range(n)]
    
    partials = partial_derivatives(poly_str, vars, x_values)
    I = construct_ideal(partials)
    reg_I = castelnuovo_mumford_regularity(I, n)

    return {
        "metric_name": metric_name,
        "metric_value": metric_value,
        "instances_tested": instances_tested,
        "conjecture_holds": conjecture_holds,
        "counterexample": counterexample
    }

if __name__ == "__main__":
