import random

def generate_tseitin_formula(n):
    if n == 1:
        return "A"
    else:
        A = "A" + str(n)
        B = "B" + str(n)
        C = "C" + str(n)
        formula = f"(~{A} & {B}) | (~{B} & {C})"
        for i in range(1, n):
            sub_formula = generate_tseitin_formula(i)
            formula = f"({formula} & ~{sub_formula})"
        return formula

def frege_proof_depth(formula):
    if formula == "A":
        return 1
    elif formula.startswith("(") and formula.endswith(")"):
        parts = formula[1:-1].split("&")
        max_depth = 0
        for part in parts:
            depth = frege_proof_depth(part)
            if depth > max_depth:
                max_depth = depth
        return 1 + max_depth
    else:
        return float('inf')
