def generate_projective_plane(q):
    if q < 2:
        raise ValueError("q must be at least 2")
    
    points = list(range(q**2 + q + 1))
    lines = []
    
    for i in range(q**2 + q + 1):
        line = [i]
        for j in range(1, q + 1):
            line.append((i * j) % (q**2 + q + 1))
        lines.append(line)
    
    return points, lines

def adjacency_matrix(points, lines):
    n = len(points)
    adj_matrix = [[0] * n for _ in range(n)]
    
    for point in points:
        for line in lines:
            if point in line:
                for other_point in line:
                    if other_point != point:
                        adj_matrix[points.index(point)][points.index(other_point)] = 1
    
    return adj_matrix

def is_connected(matrix):
    n = len(matrix)
    visited = [False] * n
    stack = [0]
    
    while stack:
        node = stack.pop()
        if not visited[node]:
            visited[node] = True
            for neighbor in range(n):
                if matrix[node][neighbor] == 1 and not visited[neighbor]:
                    stack.append(neighbor)
    
    return all(visited)

def min_width_abp(adj_matrix):
    n = len(adj_matrix)
    dp = [[float('inf')] * (n + 1) for _ in range(n)]
    dp[0][1] = 0

        for i in range(n):
            for j in range(i + 1, n):
                if adj_matrix[i][j] == 1:
                    dp[j][k] = min(dp[j][k], dp[i][k - 1] + 1)
    
    return dp[n - 1][n]
