import random

from fractions import Fraction

def gaussian_elimination(matrix):
    rows, cols = len(matrix), len(matrix[0])
    rank = 0
    for i in range(cols):
        max_row = None
        for j in range(rank, rows):
            if matrix[j][i] != 0:
                max_row = j
                break
        if max_row is None:
            continue
        matrix[i], matrix[max_row] = matrix[max_row], matrix[i]
        pivot = Fraction(matrix[i][i])
        for j in range(cols):
            matrix[i][j] /= pivot
        for j in range(rows):
            if j != i and matrix[j][i] != 0:
                factor = -matrix[j][i]
                for k in range(cols):
                    matrix[j][k] += factor * matrix[i][k]
        rank += 1
    return rank

def algebraic_k_theory_rank(n, m):
    # Construct a random Tseitin formula and its associated matrix
    variables = set(range(1, n + 1))
    clauses = []
    for _ in range(m):
        literals = [random.choice([var, -var]) for var in random.sample(variables, 2)]
        clause = literals[0]
        for lit in literals[1:]:
            if lit > 0:
                clause += " OR "
            else:
                clause += " NOT "
            clause += str(abs(lit))
        clauses.append(clause)
    
    # Construct the matrix
    matrix = []
    for i, clause in enumerate(clauses):
        row = [0] * (2 * n + 1)
        row[2 * variables.index(int(clause.split()[0]))] = 1
        if len(clause.split()) == 4:
            row[2 * variables.index(int(clause.split()[3]))] = -1
        matrix.append(row)
    
    return gaussian_elimination(matrix)
