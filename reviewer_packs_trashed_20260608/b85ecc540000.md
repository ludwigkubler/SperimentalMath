def gaussian_elimination(A):
    n = len(A)

        # Find pivot row
        max_row = i
        for k in range(i + 1, n):
            if abs(A[k][i]) > abs(A[max_row][i]):
                max_row = k
        A[i], A[max_row] = A[max_row], A[i]
        
        # Eliminate below pivot
        factor = A[i][i]
        for j in range(i + 1, n):
            A[i][j] /= factor
