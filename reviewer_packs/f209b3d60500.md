    rows, cols = len(matrix), len(matrix[0])
    rank = 0

        max_row = None
        for j in range(i, rows):
            if matrix[j][i] != 0:
                max_row = j
                break
        
        if max_row is None:
            continue
        
        # Swap rows
        matrix[i], matrix[max_row] = matrix[max_row], matrix[i]
        
        # Eliminate below pivot
        for j in range(i + 1, rows):
            factor = -matrix[j][i] / matrix[i][i]
