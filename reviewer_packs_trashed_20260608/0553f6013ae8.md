    return True

def generate_primes(count):
    primes = []

def random_cnf(n, m):
    cnf = []

        clause = [random.choice([-i, i]) for i in range(1, n + 1)]
        cnf.append(clause)
    return cnf

def evaluate_cnf(cnf, assignment):
    for clause in cnf:
        if not any([assignment[abs(lit) - 1] == (lit > 0) for lit in clause]):
            return False
    return True

    count = 0
