import random
import math
from itertools import combinations

def generate_dnf(n: int) -> list:
    dnf = set()
    for i in range(1, n + 1):
        terms = [set() for _ in range(i)]
        for j in range(n):
            if random.choice([True, False]):
                terms[j % i].add(j)
        dnf.add(tuple(sorted(term) for term in terms))
    return list(dnf)

def max_matching_size(dnf: list) -> int:
    n = len(dnf[0])
    adj_matrix = [[0] * n for _ in range(n)]
    for clause in dnf:
        for i in range(n):
            if i not in clause:
                continue
            for j in range(i + 1, n):
                if j not in clause:
                    adj_matrix[i][j] += 1
                    adj_matrix[j][i] += 1
    
    matching = []
    visited = [False] * n
    
    def dfs(u: int) -> bool:
        for v in range(n):
            if adj_matrix[u][v] > 0 and not visited[v]:
                visited[v] = True
                if v not in matching or dfs(matching[v]):
                    matching[v] = u
                    return True
        return False
    
    for i in range(n):
        if not visited[i]:
            dfs(i)
    
    return sum(1 for x in matching if x != -1)
