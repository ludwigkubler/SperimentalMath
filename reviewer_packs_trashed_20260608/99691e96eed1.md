        n = len(f)
        if n == 0:
            return 0

        for i in range(n):
            if f[i] == 1:
                chi += (-1)**i * (n - i)
        return chi

        N = len(M)
        rows = random.sample(range(N), N // 2)
        cols = random.sample(range(N), N // 2)

            for c in cols:

                    cc += 1
        return cc / (N * N) ** c
