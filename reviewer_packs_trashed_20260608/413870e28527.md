import random
import math

def run_trial(seed: int) -> dict:
    random.seed(seed)
    
    def generate_cnf(n):
        clauses = []
        for _ in range(2**n):
            clause = [random.choice([-1, 1]) * (i + 1) for i in range(n)]
            if any(clause[i] == -clause[j] for i in range(n) for j in range(i+1, n)):
                clauses.append(tuple(sorted(clause)))
        return set(clauses)
    
    def clause_indicator_polynomial(cnf):
        n = len(next(iter(cnf), (0,))[0])
        poly = [0] * (2**n)
        for clause in cnf:
            sign = 1
            for var in clause:
                if var < 0:
                    sign *= -1
                    var = -var
                poly[1 << (var - 1)] += sign
        return poly
    
    def schur_functions(n):
        schurs = []
        for i in range(2**n):
            schur = [0] * (2**n)
            schur[i] = 1
            schurs.append(schur)
        return schurs
    
    def count_vanishing_schurs(poly, schurs):
        count = 0
        for schur in schurs:
            product = 1
            for i in range(len(poly)):
                if poly[i]:
                    product *= schur[i]
            if product == 0:
                count += 1
        return count
    
    n = random.randint(5, 40)

    poly = clause_indicator_polynomial(cnf)

        "metric_value": vanishing_count,
