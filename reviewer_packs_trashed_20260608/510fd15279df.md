def gaussian_elimination(A):
    n = len(A)
    for i in range(n):
        # Find pivot row
        max_row = i
        for j in range(i+1, n):
            if abs(A[j][i]) > abs(A[max_row][i]):
                max_row = j
        A[i], A[max_row] = A[max_row], A[i]
        
        # Eliminate below the pivot
        factor = 1 / A[i][i]

            A[i][j] *= factor
        for k in range(i+1, n):
            factor = A[k][i]

    return [row[:i+1] for row in A]
