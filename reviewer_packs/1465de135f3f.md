    def graphical_realization(clauses):

        graph = {i: [] for i in range(n)}

                if set(clauses[i]) & set(clauses[j]):
                    graph[i].append(j)
                    graph[j].append(i)
        return graph

            neighbors = graph[i]
            if not neighbors:
                continue

                matrix[i][j] = 1
                matrix[j][i] = 1
            matrix[n][i] = -1
            matrix[i][n] = -1
            rank += gaussian_elimination(matrix)
        return rank
    
    def gaussian_elimination(matrix):
