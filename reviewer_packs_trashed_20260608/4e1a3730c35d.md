        rank = 0
        A = [row[:] for row in M]
        
        for i in range(min(m, n)):
            if A[i][i] == 0:
                swap_found = False
                for j in range(i + 1, m):

                        A[i], A[j] = A[j], A[i]
                        swap_found = True
                        break

                A[i][j] /= pivot

                    for k in range(n):
                        A[j][k] -= factor * A[i][k]
