        if n == 1:

            return

                if len(left) + len(right) <= 4:
                    yield left + right
    
    def evaluate(abp):
        if not abp:
            return 0
        elif len(abp) == 1:
            return abp[0]
        else:
            mid = len(abp) // 2
            return (evaluate(abp[:mid]) + evaluate(abp[mid:])) % 2
    
    def is_parity_function(abp):
        for i in range(1 << n):
            if evaluate([i] * n) != parity(i):
                return False
        return True

        x ^= x >> 2
        x ^= x >> 4
        x ^= x >> 8
        x ^= x >> 16
        return x & 1

        abp = random.choice(abps)
