import random
import math

def gaussian_elimination(A):
    n = len(A)
    for i in range(n):
        # Find pivot in column i
        max_row = i
        for j in range(i+1, n):
            if abs(A[j][i]) > abs(A[max_row][i]):
                max_row = j
        A[i], A[max_row] = A[max_row], A[i]
        
        # Eliminate entries below pivot
        factor = A[i][i]
        for j in range(i+1, n):
            A[j][i] /= factor
    
    rank = 0
    for row in A:
        if any(row):
            rank += 1
    return rank

def matrix_rank(M):
    A = [row[:] for row in M]
