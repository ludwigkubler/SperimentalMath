                continue
            clauses.append(clause)

    def resolution_width(clauses):
        queue = list(clauses)
        seen = set()
        while queue:
            literal, rest = queue.pop(0)

                        return len(queue) + 1

                    new_clauses.append(clause)
            queue.extend(new_clauses)
