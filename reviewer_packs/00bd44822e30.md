def characteristic_polynomial(cnf):

    p[0][0] = 1
    x = [Fraction(1), Fraction(-1)]

        term = Fraction(1)
        for lit in clause:
            if lit > 0:
                term *= (1 - x[-lit])
            else:
                term *= (1 + x[-lit])

            for j in range(i, -1, -1):
                p[i][j] += term * (-1) ** (i - j)
