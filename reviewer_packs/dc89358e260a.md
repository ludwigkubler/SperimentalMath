def generate_random_circuit(n, m):
    if n <= 0 or m <= 0:
        return None
    circuit = []

        gate_type = random.randint(1, 3)  # 1: AND, 2: OR, 3: NOT
        if gate_type == 3:
            inputs = [random.randint(0, 1)]
        else:
            inputs = [random.randint(0, 1) for _ in range(gate_type - 1)]
        circuit.append((gate_type, inputs))
    return circuit

def evaluate_circuit(circuit):
    if not circuit:
        return None
    stack = []
    for gate_type, inputs in reversed(circuit):
        if gate_type == 3:  # NOT
            result = 1 - inputs[0]
        else:
            if len(inputs) != gate_type - 1:
                return None
            if gate_type == 1:  # AND
                result = all(inputs)
            elif gate_type == 2:  # OR
                result = any(inputs)
            else:
                return None
        stack.append(result)
    return stack.pop()
