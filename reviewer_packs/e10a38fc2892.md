def matrix_multiply(A, B):

def gaussian_elimination(A, b):
    n = len(A)
    M = [A[i] + [b[i]] for i in range(n)]
    for i in range(n):
        max_row = i
        for j in range(i+1, n):
            if abs(M[j][i]) > abs(M[max_row][i]):

                for k in range(n + 1):

def poly_to_matrix(poly, n):

    return A
