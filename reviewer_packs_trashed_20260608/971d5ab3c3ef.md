def fourier_transform(f, n):
    result = [0] * (1 << n)
    for S in range(1 << n):
        sum_val = 0
        for x in range(1 << n):
            term = f(x) * math.exp(-2j * math.pi * S * x / (1 << n))
            sum_val += term
        result[S] = sum_val / math.sqrt(1 << n)
    return result

if __name__ == "__main__":
