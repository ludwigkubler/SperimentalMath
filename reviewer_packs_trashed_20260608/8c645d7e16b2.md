import math

def run_trial(seed: int) -> dict:

    def generate_monotone_boolean_function(n):

    def schur_polynomial(f):
        n = len(f)
        if n == 1:
            return f[0]
        else:
            s = [0] * (n + 1)
            s[0] = 1
            for i in range(n):
                s[i+1] = sum(s[j] * f[i-j] for j in range(i+1))
            return s
    
    def min_rank(poly):
        n = len(poly) - 1
        rank = 0
        for i in range(1, n + 1):
            if poly[i] != 0:
                rank += 1
        return rank
