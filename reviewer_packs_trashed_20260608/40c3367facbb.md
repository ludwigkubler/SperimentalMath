def lcm(a, b):

def matrix_multiplication(A, B):
    if len(A[0]) != len(B):
        raise ValueError("Matrix multiplication not possible")
    
    rows_A = len(A)
    cols_A = len(A[0])
    cols_B = len(B[0])

    sign = Fraction(1)
