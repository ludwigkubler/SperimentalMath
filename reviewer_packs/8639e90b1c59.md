import math

def run_trial(seed: int) -> dict:
    random.seed(seed)
    
    def generate_d_regular_circuit(n, d):
        if n % d != 0:
            return None
        circuit = [[random.randint(0, 1) for _ in range(d)] for _ in range(n)]
        return circuit
    
    def compute_tropical_rank(circuit):
        n = len(circuit)
        m = len(circuit[0])
        rank = 0
        for i in range(m):
            row_sums = [sum(row[i] for row in circuit) for row in circuit]
            if any(x > 0 for x in row_sums):
                rank += 1
        return rank
    
    def compute_monotone_width(circuit):
        n = len(circuit)
        m = len(circuit[0])
        width = 0
        for i in range(m):
            max_sum = 0
            current_sum = 0
            for j in range(n):
                if circuit[j][i] == 1:
                    current_sum += 1
                else:
                    max_sum = max(max_sum, current_sum)
                    current_sum = 0
            max_sum = max(max_sum, current_sum)
            width = max(width, max_sum)
        return width
