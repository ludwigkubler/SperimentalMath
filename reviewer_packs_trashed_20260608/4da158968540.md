def run_trial(seed: int) -> dict:

        if 2**n != len(f):
            raise ValueError("Invalid function length")

        return sum(f[i] == f[j] for i in range(n) for j in range(i+1, n)) / (n * (n - 1))

        n = int(math.log2(len(f)))

            raise ValueError("Invalid function length")
