    if len(inputs) == 1:
        return inputs[0]
    else:
        mid = len(inputs) // 2
        left_result = ac0_circuit(inputs[:mid], n)
        right_result = ac0_circuit(inputs[mid:], n)
        return (left_result + right_result) % 2
