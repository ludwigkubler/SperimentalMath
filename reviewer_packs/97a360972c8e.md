def gaussian_elimination(A):
    m, n = len(A), len(A[0])
    for i in range(m):
        max_row = i
        for j in range(i+1, m):
            if abs(A[j][i]) > abs(A[max_row][i]):
                max_row = j
        A[i], A[max_row] = A[max_row], A[i]

def compute_minimal_rank(circuit):
    n = len(circuit)
    Q = [[0] * (n + 1) for _ in range(n + 1)]
