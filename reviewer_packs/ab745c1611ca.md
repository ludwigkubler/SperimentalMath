        n = len(f)
        max_width = 0
        for i in range(1 << n):
            width = 0
            stack = []

                if (i >> j) & 1:
                    stack.append(f[j])
                    width += 1
                else:
                    if not stack:
                        continue
                    stack.pop()
                    width -= 1
            max_width = max(max_width, width)
