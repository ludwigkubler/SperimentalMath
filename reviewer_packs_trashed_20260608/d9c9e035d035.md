def gaussian_elimination(matrix):
    n = len(matrix)
    m = len(matrix[0])
    
    # Forward elimination
    for i in range(n):
        max_row = i
        for j in range(i+1, n):
            if abs(matrix[j][i]) > abs(matrix[max_row][i]):
                max_row = j
        matrix[i], matrix[max_row] = matrix[max_row], matrix[i]
        
        if matrix[i][i] == 0:
            raise ValueError("Matrix is singular")
