        clause = random.sample(literals, 3)
        clauses.append(clause)

def r_transform(matrix):
    n = len(matrix)
    if n == 0:
        return 0
    det = matrix[0][0]

            minor = [row[:j] + row[j+1:] for row in matrix[:i] + matrix[i+1:]]
