        def dpll(cnf, assignment):

                literal, _ = unit_clauses[0]

                return 1 + dpll([c for c in cnf if literal not in c], new_assignment)
            pure_literals = {}

                pos_count, neg_count = sum(1 for c in cnf if lit in c), sum(1 for c in cnf if -lit in c)
