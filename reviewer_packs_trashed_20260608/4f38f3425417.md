def gaussian_elimination(A, b):
    n = len(A)
    for i in range(n):
        # Find max pivot in column i
        max_row = i
        for j in range(i+1, n):
            if abs(A[j][i]) > abs(A[max_row][i]):
                max_row = j
        A[i], A[max_row] = A[max_row], A[i]
        b[i], b[max_row] = b[max_row], b[i]
        
        # Eliminate entries below pivot
        for j in range(i+1, n):
            factor = A[j][i] / A[i][i]
            for k in range(i, n):
                A[j][k] -= factor * A[i][k]
            b[j] -= factor * b[i]
    return [A[i][i] for i in range(n)], b

def free_cumulant_transform(M):
    n = len(M)
    M_bar = [[0] * (n+1) for _ in range(n+1)]
    for i in range(n):
        for j in range(n):
            M_bar[i][j] = M[i][j]
            M_bar[i][n] += M[i][j]
            M_bar[n][i] += M[j][i]
    M_bar[n][n] = n
    
    A_bar = [[M_bar[i][j] for j in range(n+1)] for i in range(n+1)]
    b = [0] * (n+1)
    
    det, _ = gaussian_elimination(A_bar, b)
    return det
