def gaussian_elimination(matrix):

        i_max = rank
        for i in range(rank, rows):

                i_max = i

def matrix_multiplication(A, B):
    rows_A, cols_A = len(A), len(A[0])

    C = [[0 for _ in range(cols_B)] for _ in range(rows_A)]
