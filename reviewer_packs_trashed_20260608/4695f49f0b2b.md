        n = len(f)

            return 0

        for i in range(n):
            f_i = lambda x: f(x ^ (1 << i))
            cc_i = communication_complexity(f_i)
            if cc_i < cc:
                cc = cc_i
        return cc + 1

        n = len(t_f)

            return 0

            t_f_i = [t_f[j] ^ t_f[i] for j in range(n)]
            rank_i = tropical_cycle_rank(t_f_i)
            if rank_i > rank:
                rank = rank_i
