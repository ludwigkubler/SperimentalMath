    def is_real_stable(poly):
        roots = [complex(root) for root in poly.roots]
        for root in roots:
            if not (root.imag == 0 and root.real >= 0):
                return False
        return True

            x = [x_i - step_size * grad_i for x_i, grad_i in zip(x, gradient)]
        return max(abs(x_i) for x_i in x)
