def run_trial(seed: int) -> dict:

        if n == 0 or k == 0:
            return 1

            order *= (n - i + 1) / i
        return int(math.ceil(order))
    
    def generate_k_cnf(k, n):
        cnf = []
        for _ in range(n):
            clause = [random.choice([-1, 1]) * random.randint(1, k) for _ in range(k)]
            cnf.append(clause)
        return cnf
    
    def calculate_order(cnf):
        orders = []
        for clause in cnf:
            n = len(clause)
            order = hypergeometric_order(n, k)
            orders.append(order)
