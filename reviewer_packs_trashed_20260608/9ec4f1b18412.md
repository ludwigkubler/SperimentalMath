def walsh_fourier_coefficient(F, S):
    n = len(F)

        term = (-1)**sum((i >> j) & 1 for j in range(n) if (j + 1) in S)
        sum_term += F[i] * term
    return Fraction(sum_term, 2**n)

    n = len(F)
