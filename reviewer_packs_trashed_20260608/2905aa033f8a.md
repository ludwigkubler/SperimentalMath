def find_fundamental_group_rank(graph):
    n = len(graph)
    visited = [False] * n
    rank = 0

        nonlocal rank
        visited[node] = True
        for neighbor in graph[node]:
            if not visited[neighbor]:
                dfs(neighbor, node)
            elif neighbor != parent:
                rank += 1

        if not visited[i]:
