    return clique_edges + remaining_edges

def compute_quotient_algebra_rank(edges):
    n = len(set(u for u, v in edges) | set(v for u, v in edges))
    algebra = [[0] * n for _ in range(n)]

        algebra[u][v] = 1
        algebra[v][u] = 1
