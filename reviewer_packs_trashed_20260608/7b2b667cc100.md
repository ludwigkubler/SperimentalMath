        n = int(math.log2(len(f)))
        if n == 0:
            return 0
        A = [[0] * (n + 1) for _ in range(n + 1)]
        b = [0] * (n + 1)
        for i in range(1, n + 1):
            for j in range(i + 1):
                A[i][j] = math.comb(i, j)
            b[i] = sum(f[j] for j in range(2**i))

            if A[i][i] == 0:
                return float('inf')

                factor = A[j][i] / A[i][i]
                for k in range(i, n + 1):
                    A[j][k] -= factor * A[i][k]
                b[j] -= factor * b[i]
        
        rank = 0
        for i in range(n + 1):
