import random
import math

def gaussian_elimination(A):
    m, n = len(A), len(A[0])
    rank = 0
    for j in range(n):
        pivot_row = -1
        for i in range(rank, m):
            if A[i][j] == 1:
                pivot_row = i
                break
        if pivot_row != -1:
            A[pivot_row], A[rank] = A[rank], A[pivot_row]
            rank += 1
            for i in range(m):
                if i != rank - 1 and A[i][j] == 1:
                    for k in range(n):
                        A[i][k] ^= A[rank - 1][k]
    return rank
