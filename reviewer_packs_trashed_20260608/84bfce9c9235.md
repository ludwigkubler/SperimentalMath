def generate_k_clique(n, k):
    if n < k:
        return None

    while len(clique) < n and remaining_nodes:
        v = random.choice(remaining_nodes)
        if all(v in clique for u in clique):
            clique.add(v)
        remaining_nodes.remove(v)
    
    return clique
