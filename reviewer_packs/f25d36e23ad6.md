import random
import math
from fractions import Fraction

def generate_boolean_function(n):
    return [random.randint(0, 1) for _ in range(2**n)]

def hodge_rank(f):
    n = int(math.log2(len(f)))
    if len(f) != 2**n:
        raise ValueError("Input must be a valid n-bit boolean function")
    
    count_0 = [0] * n
    count_1 = [0] * n
    
    for x in f:
        for i in range(n):
            if x & (1 << i) == 0:
                count_0[i] += 1
            else:
                count_1[i] += 1
    
    rank = max(max(count_0), max(count_1))
    return rank

def run_trial(seed: int) -> dict:
    random.seed(seed)
    
    n_values = [5, 10, 15, 20, 30, 40]
    results = []
    
    for n in n_values:
        instances_tested = 0
        total_rank = 0
        
        while instances_tested < 30:
            f = generate_boolean_function(n)
            rank = hodge_rank(f)
            total_rank += rank
            instances_tested += 1
        
        avg_rank = Fraction(total_rank, instances_tested)
        results.append(avg_rank)
    
    mean_value = sum(results) / len(results)
    std_dev = math.sqrt(sum((x - mean_value)**2 for x in results) / len(results))

        "metric_value": float(mean_value),

        "conjecture_holds": conjecture_holds,
        "counterexample": counterexample
    }
