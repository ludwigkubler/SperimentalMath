    def communication_complexity(f):
        n = len(f)
        max_communication = 0
        for i in range(2**n):
            comm = sum(f[j] != f[i ^ j] for j in range(n))
            if comm > max_communication:
                max_communication = comm
        return max_communication
    
    def compute_k_group_rank(f):
        n = len(f)
        k_group_rank = 0
        for i in range(2**n):

            if subset_sum == 0:
                k_group_rank += 1

    def linear_regression(x, y):
        n = len(x)
        mean_x = sum(x) / n
