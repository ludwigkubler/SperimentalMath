import random
import math

def generate_disjointness_matrix(n):
    M = [[0] * n for _ in range(n)]
    for i in range(n):
        for j in range(i + 1, n):
            if random.choice([True, False]):
                M[i][j] = 1
                M[j][i] = 1
    return M

def matrix_multiplication(A, B):
    n = len(A)
    C = [[0] * n for _ in range(n)]
    for i in range(n):
        for j in range(n):
            for k in range(n):
                C[i][j] += A[i][k] * B[k][j]
    return C

def trace(matrix):
    n = len(matrix)
    tr = 0
    for i in range(n):
        tr += matrix[i][i]
    return tr

def schatten_p_norm(M, p):
    n = len(M)
    A = matrix_multiplication(M, M)
    det_A = 1.0
    for i in range(n):
        det_A *= A[i][i]
    if det_A <= 0:
        return float('inf')
    return (det_A ** (1 / p)) * n ** (-1 / p)
