def run_trial(seed: int) -> dict:

        return [f'NOT {left[0]}', f'{right[0]} OR {left[1]}']

    def hodge_rank(tree):
        if isinstance(tree, str):
            return 1
        else:
            return max(hodge_rank(subtree) for subtree in tree)

    results = []

        results.append(rank)
