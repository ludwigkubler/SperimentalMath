def matrix_multiplication(A, B):
    n = len(A)
    C = [[0 for _ in range(n)] for _ in range(n)]

        for j in range(n):
            for k in range(n):
                C[i][j] += A[i][k] * B[k][j]
    return C

def identity_matrix(n):
    I = [[0 for _ in range(n)] for _ in range(n)]
    for i in range(n):
        I[i][i] = 1
    return I

def inverse(A):
    n = len(A)
    I = identity_matrix(n)
    for k in range(n):
        pivot = A[k][k]
        if pivot == 0:
            raise ValueError("Matrix is singular")
        for j in range(n):
            A[k][j] /= pivot
            I[k][j] /= pivot
        for i in range(n):
            if i != k:
                factor = A[i][k]
                for j in range(n):
                    A[i][j] -= factor * A[k][j]
                    I[i][j] -= factor * I[k][j]
    return I

def eigenvalues(A):
    n = len(A)
    lambda_values = []
    for _ in range(10):  # Simple power iteration method
        x = [random.random() for _ in range(n)]
        x_norm = sum(x[i] ** 2 for i in range(n)) ** 0.5
        x = [x_i / x_norm for x_i in x]
        for _ in range(10):
            y = matrix_multiplication(A, x)
            y_norm = sum(y[i] ** 2 for i in range(n)) ** 0.5
            y = [y_i / y_norm for y_i in y]
            lambda_new = sum(x[i] * y[i] for i in range(n))
            if abs(lambda_new - lambda_values[-1]) < 1e-6:
