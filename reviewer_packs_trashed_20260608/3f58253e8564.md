                tseitin_count += 1

            tseitin_clauses.append(tseitin_clause)

        def resolve(clauses):
            resolved = set()
            while True:
                new_resolved = False

                        if any(-x in clauses[i] and x in clauses[j] for x in set(clauses[i]) & set(clauses[j])):

                    break
