def factorial(n):
    if n == 0:
        return 1
    result = 1
    for i in range(2, n + 1):
        result *= i
    return result

def hook_length_formula(matrix, shape):
    n = len(shape)
    numerator = 1
    denominator = 1
    for row in range(n):

            denominator *= matrix[row][col]
    return Fraction(numerator, denominator)
