        return clauses
    
    def resolution_width(cnf):
        queue = cnf[:]
        seen = set()
        while queue:
            literal = random.choice(queue)
            if literal in seen or -literal in seen:
                return len(seen)
            seen.add(literal)

                    leaf.append(random.choice([-1, 1]) * (j + 1))
            leaves.add(tuple(sorted(leaf)))
