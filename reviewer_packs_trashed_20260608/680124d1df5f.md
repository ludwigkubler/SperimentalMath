def association_scheme(cnf):
    n = max(max(clause) for clause in cnf)
    adjacency_matrix = [[0] * (n + 1) for _ in range(n + 1)]

            if literal > 0:
                adjacency_matrix[literal][literal] += 1
            else:
                neg_literal = -literal
