def fourier_coefficients(f, n):
    coeffs = [0] * (2**n)
    for x in range(2**n):
        sum_val = 0
        for y in range(2**n):
            term = f(y) * (-1)**(x & y)
            sum_val += term
        coeffs[x] = sum_val / (2**n)
    return coeffs

def algebraic_k_theory_rank(coeffs):
    n = len(coeffs)
    rank = 0
    for i in range(n):
