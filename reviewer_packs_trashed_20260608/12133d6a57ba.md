def run_trial(seed: int) -> dict:

    def communication_rank(f):
        n = int(math.log2(len(f)))
        count = Counter()
        for i in range(2**n):
            for j in range(i+1, 2**n):
                if f[i] != f[j]:
                    count[(i, j)] += 1
        return len(count)
    
    def geometric_complexity(f):
        n = int(math.log2(len(f)))
        count = Counter()
        total = 0
        for i in range(2**n):
            for j in range(i+1, 2**n):
                if f[i] != f[j]:
                    count[(i, j)] += 1
                    total += 1
        return sum(count[i] * math.log2((count[i]+1)/total) for i in count)
