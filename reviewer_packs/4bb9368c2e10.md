        q = [[0] * (n + 1) for _ in range(n + 1)]
        for i in range(n + 1):
            for j in range(n + 1):
                if (i << n) | (1 << j) - 1 < len(f):
                    q[i][j] = f[(i << n) | (1 << j) - 1]
        return q

        m, n = len(matrix), len(matrix[0])

            if matrix[i][i] == 0:
                found_pivot = False
                for k in range(i + 1, m):
                    if matrix[k][i] != 0:
                        matrix[i], matrix[k] = matrix[k], matrix[i]
                        found_pivot = True
                        break

                matrix[i][j] /= pivot
