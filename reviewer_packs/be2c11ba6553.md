        return result
    
    def is_prime(n):
        if n <= 1:
            return False
        if n <= 3:
            return True
        if n % 2 == 0 or n % 3 == 0:
            return False

                return False
            i += 6
        return True

        return [random.choice([0, 1]) for _ in range(2 ** n)]

            return None

        for i in range(n):
