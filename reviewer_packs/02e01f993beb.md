def compute_monotone_width(circuit):
    n = len(circuit)
    width = 0
    for i in range(1 << n):
        active = [j for j in range(n) if (i >> j) & 1]
        if all(circuit[j] == circuit[active[0]] for j in active):
            width = max(width, len(active))
    return width
