def gaussian_elimination(A, b):
    m, n = len(A), len(b)
    for i in range(m):
        max_row = i
        for j in range(i + 1, m):
            if abs(A[j][i]) > abs(A[max_row][i]):
                max_row = j
        A[i], A[max_row] = A[max_row], A[i]

            b[j] -= factor * b[i]

def determinant(A):
