        n = len(A)

            max_row = i + max(range(i, n), key=lambda j: abs(A[j][i]))
            A[i], A[max_row] = A[max_row], A[i]
            for j in range(i+1, n):
                factor = A[j][i] / A[i][i]
                for k in range(n):
                    A[j][k] -= factor * A[i][k]
        return A
    
    def matrix_multiply(A, B):
        n = len(A)
        C = [[0] * n for _ in range(n)]
        for i in range(n):
            for j in range(n):
                for k in range(n):
                    C[i][j] += A[i][k] * B[k][j]
        return C
    
    def moment_cumulant_transform(A):
        n = len(A)
        cumulants = [0] * (n + 1)
        cumulants[0] = 1

            for j in range(i, n):
                factor = A[i][j]
                for k in range(j+1, n):
                    factor *= A[j][k]
                cumulants[i+1] += factor
        return cumulants
    
    def free_cumulant_gap(cumulants):
        gap = 0

        return gap
