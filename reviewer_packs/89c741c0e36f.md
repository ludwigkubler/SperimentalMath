        clause = [random.randint(1, n) * (-1 if random.choice([True, False]) else 1)
                  for _ in range(random.randint(1, n))]

def integer_to_monomial(i, n):
    monomial = ""
    for j in range(n):
        if i & (1 << j):
            monomial += f"x{j+1}"
        elif i & (1 << (-j - 1)):
            monomial += f"~x{j+1}"
    return monomial

def generate_monomial_ideal(cnf, n):
    ideal = set()
