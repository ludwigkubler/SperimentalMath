    def gaussian_elimination(A):
        m, n = len(A), len(A[0])
        for i in range(m):
            max_row = i
            for j in range(i+1, m):
                if abs(A[j][i]) > abs(A[max_row][i]):
                    max_row = j
            A[i], A[max_row] = A[max_row], A[i]
            if A[i][i] == 0:

                A[i][j] /= A[i][i]

                    for j in range(n):
                        A[k][j] -= factor * A[i][j]

        return math.sqrt(variance)
