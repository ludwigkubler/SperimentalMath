    def is_independent(lattice, matrix):
        for i in range(len(matrix)):
            if any(all(matrix[i][j] == lattice[j][k] for k in range(len(matrix))) for j in range(i + 1, len(matrix))):
                return False
        return True

        lattices = []

                    lattices.append(lattice)
