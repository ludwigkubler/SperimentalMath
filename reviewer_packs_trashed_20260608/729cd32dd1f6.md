def generate_boolean_function(n):
    return [random.choice([0, 1]) for _ in range(2**n)]

def formal_group_rank(f):
    n = int(math.log2(len(f)))
    A = [[0] * (n + 1) for _ in range(n + 1)]
    for i in range(n + 1):
        for j in range(n + 1):
            if i == 0 and j == 0:
                A[i][j] = 1
            elif i == 0 or j == 0:
                A[i][j] = 0
            else:
                A[i][j] = f[2**(i-1) + 2**(j-1)]
    rank = 0
    for row in A:
        if any(row):
            rank += 1
    return rank

def read_twice_bp_size(f):
    n = int(math.log2(len(f)))
    size = 1
    for i in range(n + 1):
        if f[2**i]:
            size += 1
    return size
