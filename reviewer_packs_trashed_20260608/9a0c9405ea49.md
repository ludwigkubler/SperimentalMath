def gaussian_elimination(A):
    m, n = len(A), len(A[0])
    rank = 0

        pivot = -1

            if A[j][i] != 0:
                pivot = j
                break

            continue
        A[pivot], A[rank] = A[rank], A[pivot]

    return rank
