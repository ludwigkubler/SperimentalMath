        n = len(instance)
        trop_points = []
        for i in range(2**n):
            point = []
            for j in range(n):
                if instance[i] & (1 << j):
                    point.append(j + 1)
                else:
                    point.append(-j - 1)
            trop_points.append(point)

        n = len(trop_points[0])

                symmetries.append(i)
